cmake_minimum_required(VERSION 3.25)
project(FlashFasaFiso VERSION 1.0.0 LANGUAGES CXX RC)

if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8
   OR NOT CMAKE_CXX_COMPILER_ARCHITECTURE_ID STREQUAL "x64")
  message(FATAL_ERROR "FlashFasaFiso requires Windows x64 and MSVC.")
endif()

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(FFF_ASAN "Enable MSVC AddressSanitizer" OFF)
option(FFF_ANALYZE "Enable MSVC static analysis" OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
add_compile_options($<$<COMPILE_LANGUAGE:CXX>:/W4> $<$<COMPILE_LANGUAGE:CXX>:/utf-8>
  $<$<COMPILE_LANGUAGE:CXX>:/permissive-> $<$<COMPILE_LANGUAGE:CXX>:/Zc:__cplusplus>)
if(FFF_ASAN)
  add_compile_options($<$<COMPILE_LANGUAGE:CXX>:/fsanitize=address> $<$<COMPILE_LANGUAGE:CXX>:/Zi>)
  add_link_options(/INCREMENTAL:NO /DEBUG)
endif()
if(FFF_ANALYZE)
  add_compile_options($<$<COMPILE_LANGUAGE:CXX>:/analyze>)
endif()

add_library(fff_core STATIC src/library.cpp src/ruffle.cpp)
target_include_directories(fff_core PUBLIC src)
target_link_libraries(fff_core PUBLIC bcrypt)

set(fffWindowSources src/window.cpp src/worker.cpp src/app.rc)
set(fffWindowLibraries fff_core d2d1 dwrite windowscodecs dwmapi comctl32 shell32 ole32 gdi32)
add_executable(FlashFasaFiso WIN32 src/main.cpp ${fffWindowSources})
target_link_options(FlashFasaFiso PRIVATE /MANIFEST:NO)
target_link_libraries(FlashFasaFiso PRIVATE ${fffWindowLibraries})

include(CTest)
if(BUILD_TESTING)
  add_executable(fff_dpi_smoke EXCLUDE_FROM_ALL tests/dpi_smoke.cpp ${fffWindowSources})
  target_link_options(fff_dpi_smoke PRIVATE /MANIFEST:NO)
  target_link_libraries(fff_dpi_smoke PRIVATE ${fffWindowLibraries})
  add_executable(fff_tests tests/library_tests.cpp)
  target_link_libraries(fff_tests PRIVATE fff_core shell32)
  add_test(NAME Library COMMAND fff_tests)
  set_tests_properties(Library PROPERTIES TIMEOUT 20)
  if(FFF_ASAN)
    get_filename_component(fffCompilerDir "${CMAKE_CXX_COMPILER}" DIRECTORY)
    set_tests_properties(Library PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${fffCompilerDir}")
  endif()
endif()
